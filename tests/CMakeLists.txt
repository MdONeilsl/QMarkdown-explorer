# -----------------------------------------------------------------------------
#  Test suite for QMarkdown-explorer.
#
#  Each test is a standalone executable that compiles only the sources it
#  exercises.  Building a test does not require building the application
#  binary, and the whole suite is discovered by `ctest` from the build
#  directory:
#
#      ctest --output-on-failure -C Debug
#
#  The environment variable QT_QPA_PLATFORM=offscreen is set per-test so
#  the suite runs on a headless CI runner without X or a Windows console
#  session.  The offscreen plugin is bundled with Qt 6 on every platform
#  and requires no display.
#
#  Include paths:
#    * ${CMAKE_SOURCE_DIR}/src — every test includes at least one project
#      header by bare name (`#include "RepoList.h"`).  The test sources
#      live in tests/, not src/, so the implicit same-directory search
#      does not find them; the src/ directory must be on the include
#      path explicitly.
#    * ${MD_GENERATED_INCLUDE} — config.h, generated by CMake from
#      src/config.h.in.
#
#  DLL deployment (Windows):
#
#  Every test links Qt6::Test, which the application does not — so
#  Qt6Test[d].dll must be next to each test binary.  Beyond that, tests
#  that construct a QApplication (QTEST_MAIN) also require the platform
#  plugin, which is loaded lazily when QApplication is created.  Without
#  platforms/qwindows[d].dll the process aborts at startup with:
#
#      "This application failed to start because no Qt platform plugin
#       could be initialized."
#
#  add_test_qt_runtime() below copies the four required DLLs and the
#  platform plugin folder next to every test binary.
#
#  Note: because this step runs as a POST_BUILD per test target, running
#  a parallel build (`cmake --build … --parallel`) can hit
#  "Permission denied" while several targets attempt to update the
#  shared platforms/ folder at the same time.  A single shared deploy
#  target would eliminate the race but is not currently in place.
# -----------------------------------------------------------------------------

set(MD_GENERATED_INCLUDE ${CMAKE_BINARY_DIR}/generated)

# Convenience: the Qt modules every GUI-adjacent test needs.
set(MD_TEST_LIBS Qt6::Test Qt6::Widgets)

# -----------------------------------------------------------------------------
#  add_test_qt_runtime(<target>)
#
#  Copies the Qt runtime DLLs and the platform plugin next to <target>
#  on Windows.  No-op on every other platform.
# -----------------------------------------------------------------------------
function(add_test_qt_runtime target)
    if(NOT WIN32)
        return()
    endif()
    if(NOT TARGET Qt6::Test)
        return()
    endif()

    add_custom_command(TARGET ${target} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            $<TARGET_FILE:Qt6::Core>
            $<TARGET_FILE:Qt6::Gui>
            $<TARGET_FILE:Qt6::Widgets>
            $<TARGET_FILE:Qt6::Test>
            $<TARGET_FILE_DIR:${target}>
        COMMENT "Copying Qt runtime DLLs next to ${target}"
        VERBATIM
    )

    if(DEFINED QT6_INSTALL_PREFIX
       AND EXISTS "${QT6_INSTALL_PREFIX}/plugins/platforms")
        add_custom_command(TARGET ${target} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E make_directory
                "$<TARGET_FILE_DIR:${target}>/platforms"
            COMMAND ${CMAKE_COMMAND} -E copy_directory
                "${QT6_INSTALL_PREFIX}/plugins/platforms"
                "$<TARGET_FILE_DIR:${target}>/platforms"
            COMMENT "Copying Qt platform plugins next to ${target}"
            VERBATIM
        )
    else()
        message(WARNING
            "Cannot locate the Qt platform plugin directory for target "
            "'${target}'.  '${QT6_INSTALL_PREFIX}/plugins/platforms' does "
            "not exist.  Tests that create a QApplication will fail to "
            "start.")
    endif()
endfunction()

#  tst_imageresolver
# -----------------------------------------------------------------------------
qt_add_executable(tst_imageresolver
    tst_imageresolver.cpp
    ${CMAKE_SOURCE_DIR}/src/ImageResolver.cpp
    ${CMAKE_SOURCE_DIR}/src/ImageResolver.h
)
target_include_directories(tst_imageresolver PRIVATE
    ${CMAKE_SOURCE_DIR}/src
    ${MD_GENERATED_INCLUDE}
)
target_link_libraries(tst_imageresolver PRIVATE ${MD_TEST_LIBS})
add_test_qt_runtime(tst_imageresolver)
add_test(NAME tst_imageresolver COMMAND tst_imageresolver)
set_tests_properties(tst_imageresolver PROPERTIES
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen")

# -----------------------------------------------------------------------------
#  tst_cli — end-to-end command-line tests via QProcess.
#
#  The application binary path is passed via an environment variable
#  (QMD_APP_PATH) rather than as argv[1].  QTEST_GUILESS_MAIN parses argv
#  looking for test function names; a positional argument like
#  "C:/…/QMarkdown-explorer.exe" is interpreted as a test function called
#  "C", and the run fails before any of the test slots execute:
#
#      Unknown test function: 'C'
#
#  Using ${CMAKE_COMMAND} -E env sidesteps argv entirely.
# -----------------------------------------------------------------------------
qt_add_executable(tst_cli
    tst_cli.cpp
)
target_link_libraries(tst_cli PRIVATE Qt6::Test Qt6::Core)
add_test_qt_runtime(tst_cli)
add_dependencies(tst_cli QMarkdown-explorer)
add_test(NAME tst_cli COMMAND
    ${CMAKE_COMMAND} -E env
    "QT_QPA_PLATFORM=offscreen"
    "QMD_APP_PATH=$<TARGET_FILE:QMarkdown-explorer>"
    $<TARGET_FILE:tst_cli>)
